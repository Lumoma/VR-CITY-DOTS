94 private bool FindReferencesInScene()

118 private IEnumerator RunMeasurementLoop()
119 {
120 StringBuilder csv = new StringBuilder();
121 string sceneName = SceneManager.GetActiveScene().name;
122 csv.AppendLine("Scene;Type;AgentCount;AvgFPS;MinFPS;MaxFPS;1PercentLowFPS;AvgFrameTime_ms");
123 foreach (int count in agentCounts)
124 {
125 LogStatus($"Messe {count} Agenten ({_currentTypeString})...");
126 SetAgentCount(count);
127 yield return new WaitForSeconds(warmupTime);
128 List<float> frames = new List<float>();
129 float elapsed = 0f;
130 while (elapsed < measureTime)
131 {
132 float dt = Time.unscaledDeltaTime;
133 frames.Add(dt);
134 elapsed += dt;
135 yield return null;
136 }
137 RecordMetric(csv, sceneName, count, frames);
138 }
139 SaveFile(sceneName, csv.ToString());
140 SetAgentCount(0);
141 yield return new WaitForSeconds(1.0f);
142 }

162 private void RecordMetric(StringBuilder csv, string sceneName, int count, List<float> frameTimes)
163 {
164 if (frameTimes.Count == 0) return;
165 float avgFrameTime = frameTimes.Average();
166 float avgFPS = 1.0f / avgFrameTime;
167 float minFPS = 1.0f / frameTimes.Max();
168 float maxFPS = 1.0f / frameTimes.Min();
169 frameTimes.Sort((a, b) => b.CompareTo(a));
170 int index1Percent = Mathf.CeilToInt(frameTimes.Count * 0.01f);
171 float p1Low = 1.0f / frameTimes[Mathf.Clamp(index1Percent, 0, frameTimes.Count - 1)];
172 string line = string.Format(System.Globalization.CultureInfo.InvariantCulture,
173 "{0};{1};{2};{3:F2};{4:F2};{5:F2};{6:F2};{7:F4}",
174 sceneName, _currentTypeString, count, avgFPS, minFPS, maxFPS, p1Low, avgFrameTime * 1000f
175 );
176 csv.AppendLine(line);
177 }
