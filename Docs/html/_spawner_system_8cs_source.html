35 public void OnUpdate(ref SystemState state)
36 {
37 // 1. Spawner Entity holen
38 var spawnerEntity = SystemAPI.GetSingletonEntity<SpawnerData>();
39 var config = SystemAPI.GetComponent<SpawnerData>(spawnerEntity);
40
41 // 2. Alte Agenten löschen
42 // Wir suchen alle Entities, die den "SpawnedAgentTag" haben
43
44 //Version 1 Die einfach nur das Entity löscht
45 //var oldAgentsQuery = SystemAPI.QueryBuilder().WithAll<SpawnedAgentTag>().Build();
46 //state.EntityManager.DestroyEntity(oldAgentsQuery);
47
48 //Version 2 Die auch alle Child Objekte mit Löscht (wegen Auto GameObject)
49 var oldAgentsQuery = SystemAPI.QueryBuilder().WithAll<SpawnedAgentTag>().Build();
50 var oldAgents = oldAgentsQuery.ToEntityArray(Allocator.Temp);
51 for (int i = 0; i < oldAgents.Length; i++)
52 {
53 var entity = oldAgents[i];
54 // Optional: zusätzliche Prüfungen möglich, z.B. auf LinkedEntityGroup
55 state.EntityManager.DestroyEntity(entity);
56 }
57 oldAgents.Dispose();
58
59 // 3. Neue Agenten erstellen
60 var instances = state.EntityManager.Instantiate(config.PrefabEntity, config.Count, Allocator.Temp);
61 var random = new Unity.Mathematics.Random((uint)SystemAPI.Time.ElapsedTime + 1);
62
63 for (int i = 0; i < instances.Length; i++)
64 {
65 var entity = instances[i];
66
67 // Zufallsposition
68 float3 randomPos = config.SpawnCenter + new float3(
69 random.NextFloat(-config.AreaSize.x / 2, config.AreaSize.x / 2),
70 0,
71 random.NextFloat(-config.AreaSize.z / 2, config.AreaSize.z / 2)
72 );
73
74 quaternion randomRot = quaternion.RotateY(random.NextFloat(0, math.PI * 2));
75
76 // Transform setzen
77 var transform = LocalTransform.FromPositionRotation(randomPos, randomRot);
78 state.EntityManager.SetComponentData(entity, transform);
79
80 // RandomWalker Daten setzen
81 if (state.EntityManager.HasComponent<RandomMovementData>(entity))
82 {
83 var movementData = state.EntityManager.GetComponentData<RandomMovementData>(entity);
84 movementData.AreaSize = config.AreaSize;
85 movementData.AreaCenter = config.SpawnCenter;
86 movementData.RandomGenerator = new Unity.Mathematics.Random(random.NextUInt());
87 state.EntityManager.SetComponentData(entity, movementData);
88 }
89
90 // WICHTIG: Den Tag hinzufügen, damit wir sie beim nächsten Mal löschen können
91 state.EntityManager.AddComponent<SpawnedAgentTag>(entity);
92 }
93
94 // 4. Request entfernen
95 // Damit das System im nächsten Frame nicht wieder läuft, entfernen wir den Befehl
96 state.EntityManager.RemoveComponent<RespawnRequest>(spawnerEntity);
97 }
98 }
