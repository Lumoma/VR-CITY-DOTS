56 private void Execute(ref RandomMovementData data, ref LocalTransform transform, ref PhysicsVelocity velocity)
57 {
58 // --- 1. Rotation korrigieren (Der Zombie-Fix) ---
59 // Wir zwingen die Entity, absolut aufrecht zu stehen.
60 // Wir nehmen die aktuelle Rotation, aber löschen jegliches Kippen (X/Z).
61 float3 currentForward = math.forward(transform.Rotation);
62
63 // Wir berechnen eine neue Rotation, die nur um die Y-Achse (oben) schaut
64 // math.atan2 gibt uns den Winkel auf dem Boden
65 float currentYAngle = math.atan2(currentForward.x, currentForward.z);
66 transform.Rotation = quaternion.RotateY(currentYAngle);
67
68 // --- 2. Bewegung ---
69 float3 forward = math.forward(transform.Rotation);
70 float currentVerticalVelocity = velocity.Linear.y;
71
72 float3 newVelocity = forward * data.MovementSpeed;
73 newVelocity.y = currentVerticalVelocity; // Schwerkraft beibehalten
74
75 velocity.Linear = newVelocity;
76
77 // WICHTIG: Drehimpuls komplett töten, damit sie nicht trudeln
78 velocity.Angular = float3.zero;
79
80 // --- 3. Bounds Check ---
81 if (IsOutOfBounds(transform.Position, data.AreaCenter, data.AreaSize))
82 {
83 HandleOutOfBounds(ref data, ref transform, forward);
84 }
85 }

90 private void HandleOutOfBounds(ref RandomMovementData data, ref LocalTransform transform, float3 forward)
91 {
92 if (CurrentTime - data.LastCollisionTime < data.CooldownDuration) return;

94 data.LastCollisionTime = CurrentTime;
