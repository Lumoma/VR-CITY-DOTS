117 public void SetCountFromSlider(float sliderValue)
118 {
119 int newCount = Mathf.Max(0, Mathf.RoundToInt(sliderValue));
120 if (spawnCount == newCount) return;
121 if (newCount > spawnCount)
122 {
123 int toAdd = newCount - spawnCount;
124 AddAgents(toAdd);
125 }
126 else
127 {
128 int toRemove = spawnCount - newCount;
129 RemoveAgents(toRemove);
130 }
131 spawnCount = newCount;
132 UpdateUiText();
133 }

139 private void AddAgents(int countToAdd)
140 {
141 if (!HasValidPrefabs())
142 {
143 Debug.LogWarning("Spawner: Keine Prefabs zugewiesen! Keine Agenten hinzugefügt.");
144 return;
145 }
146 for (int i = 0; i < countToAdd; i++)
147 {

149 }
150 }

156 private void RemoveAgents(int countToRemove)
157 {
158 for (int i = 0; i < countToRemove && _spawnedAgents.Count > 0; i++)
159 {
160 int lastIndex = _spawnedAgents.Count - 1;
161 GameObject agent = _spawnedAgents[lastIndex];
162 if (agent != null) Destroy(agent);
163 _spawnedAgents.RemoveAt(lastIndex);
164 }
165 }

170 public void RespawnAllAgents()

201 private void SpawnSingleAgent(int index)
202 {
203 GameObject prefabToSpawn = SelectPrefab(index);
204 if (prefabToSpawn == null) return;
205 float radius = GetPrefabRadius(prefabToSpawn);
206 Vector3 position = TryFindSpawnPosition(radius);
207 Quaternion rotation = GetInitialRotation();
208 GameObject newAgent = Instantiate(prefabToSpawn, position, rotation);
209 SetupAgentHierarchy(newAgent, prefabToSpawn.name, index);
210 SetupRandomWalker(newAgent);
211 _spawnedAgents.Add(newAgent);
212 }

242 private Vector3 TryFindSpawnPosition(float agentRadius)
243 {
244 for (int attempt = 0; attempt < maxPlacementAttempts; attempt++)
245 {
246 Vector3 candidatePos = GetRandomPositionInArea();
247 if (Physics.OverlapSphere(candidatePos, agentRadius * 0.6f, collisionCheckLayer).Length == 0)
248 {
249 return candidatePos;
250 }
251 }
252 return GetRandomPositionInArea();
253 }
254
259 private Vector3 GetRandomPositionInArea()
260 {
261 Vector3 halfSize = areaSize * 0.5f;
262 float randomX = Random.Range(-halfSize.x, halfSize.x);
263 float randomZ = Random.Range(-halfSize.z, halfSize.z);
264 return new Vector3(transform.position.x + randomX, 0f, transform.position.z + randomZ);
265 }

271 private Quaternion GetInitialRotation()
272 {

274 return Quaternion.Euler(0f, yAngle, 0f);
275 }

282 private float GetPrefabRadius(GameObject prefab)
283 {
284 var rend = prefab.GetComponentInChildren<Renderer>();

287 return rend.bounds.extents.magnitude * agentScale;
