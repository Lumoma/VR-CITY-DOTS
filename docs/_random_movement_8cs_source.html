170 private bool IsOutOfBounds()
171 {
172 Vector3 pos = transform.position;
173 Vector3 min = areaCenter - areaSize * 0.5f;
174 Vector3 max = areaCenter + areaSize * 0.5f;
175 bool outsideX = pos.x < min.x || pos.x > max.x;
176 bool outsideY = pos.y < min.y || pos.y > max.y;
177 bool outsideZ = pos.z < min.z || pos.z > max.z;
178 return outsideX || outsideY || outsideZ;
179 }

234 private void UpdateCollider(Mesh mesh, ColliderType colliderType)
235 {
236 RemoveExistingColliders();
237 Bounds bounds = mesh != null ? mesh.bounds : new Bounds(Vector3.zero, Vector3.one);
238 Vector3 scaledCenter = Vector3.Scale(bounds.center, transform.localScale);
239 Vector3 scaledSize = Vector3.Scale(bounds.size, transform.localScale);
240 Vector3 scaledExtents = Vector3.Scale(bounds.extents, transform.localScale);
241 if (colliderType == ColliderType.Box)
242 {
243 var boxCollider = gameObject.AddComponent<BoxCollider>();
244 boxCollider.center = scaledCenter;
245 boxCollider.size = scaledSize;
246 }
247 else
248 {
249 var sphereCollider = gameObject.AddComponent<SphereCollider>();
250 sphereCollider.center = scaledCenter;
251 sphereCollider.radius = Mathf.Max(scaledExtents.x, scaledExtents.y, scaledExtents.z);
252 }
253 }
